Créer les tables dans Supabase
Allez dans votre projet Supabase → SQL Editor → New query, collez le SQL ci-dessous et cliquez Run.
create table if not exists questionnaires (
  id text primary key,
  code text unique not null,
  title text not null,
  description text,
  ref_text text,
  type text default 'mixed',
  questions jsonb default '[]',
  active boolean default true,
  max_attempts integer default 1,
  group_mode boolean default false,
  require_signature boolean default false,
  duration integer default 0,
  duration_unit text default 'minutes',
  emailjs_service text,
  emailjs_template text,
  emailjs_key text,
  created_at timestamptz default now()
);

create table if not exists responses (
  id text primary key,
  questionnaire_id text references questionnaires(id) on delete cascade,
  respondent_name text not null,
  respondent_email text not null,
  answers jsonb default '{}',
  manual_scores jsonb default '{}',
  final_score integer,
  total_points integer,
  score_pct integer,
  graded boolean default false,
  group_members jsonb default '[]',
  signature text,
  comments jsonb default '{}',
  created_at timestamptz default now()
);

alter table questionnaires enable row level security;
alter table responses enable row level security;

create policy "public_read_q" on questionnaires for select using (true);
create policy "admin_all_q" on questionnaires for all using (true);
create policy "public_insert_r" on responses for insert with check (true);
create policy "admin_read_r" on responses for select using (true);
create policy "admin_delete_r" on responses for delete using (true);
